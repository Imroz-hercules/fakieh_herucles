

################## tr_BatchMerger ##################
 create trigger [SIMATIC_BATCH].[tr_BatchMerger] on [SIMATIC_BATCH].[Batch] INSTEAD of INSERT as begin 	set NOCOUNT on; 	merge into [SIMATIC_BATCH].[Batch] as [Target] 	using ( select * from inserted ) as [Source] 	on [Target].OGUID = [Source].OGUID 	when MATCHED then update set 		[Created]					 = [Source].[Created]						, 		[ExAttribute]				 = [Source].[ExAttribute]					, 		[Quantity]					 = [Source].[Quantity]						, 		[ProductName]				= [Source].[ProductName]			, 		[ProductCode]				= [Source].[ProductCode]			, 		[ProductGUID]				= [Source].[ProductGUID]			, 		[ProductUoMId]				= [Source].[ProductUoMId]			, 		[FormulaCategoryName]		 = [Source].[FormulaCategoryName]			, 		[FormulaName]				 = [Source].[FormulaName]					, 		[FormulaVersion]			 = [Source].[FormulaVersion]				, 		[FormulaGUID]				 = [Source].[FormulaGUID]					, 		[FormulaId]					 = [Source].[FormulaId]						, 		[MRecipeName]				 = [Source].[MRecipeName]					, 		[MRecipeVersion]			 = [Source].[MRecipeVersion]				, 		[MRecipeGUID]				 = [Source].[MRecipeGUID]					,  		[MRecipeId]					 = [Source].[MRecipeId]						, 		[OrderId]					 = [Source].[OrderId]						, 		[PlanEnd]					 = [Source].[PlanEnd]						, 		[PCellGUID]					 = [Source].[PCellGUID]						, 		[PCellId]					 = [Source].[PCellId]						, 		[PartId]					 = [Source].[PartId]						, 		[BatchReleaseGroupId]	 	 = [Source].[BatchReleaseGroupId]			,	 		[CreationDateTime]		 	 = [Source].[CreationDateTime]				,	 		[BatchHasNoTags]			 = [Source].[BatchHasNoTags]				, 	 		[LastOSCDateTime]				= [Source].[LastOSCDateTime]				, 		[MarkByOSC]						= [Source].[MarkByOSC]						, 		[BatchTransferTime]			= [Source].[BatchTransferTime]		, 		[InternalSettings]			= [Source].[InternalSettings] 		 	when NOT MATCHED then insert values 	( 		[Source].[ROOTGUID]					, 		[Source].[ROOTOBJID]					, 		[Source].[ROOTOTID]					, 		[Source].[OGUID]						, 		[Source].[OBJID]						, 		[Source].[OTID]						, 		[Source].[Created]					, 		[Source].[ExAttribute]				, 		[Source].[Quantity]					, 		[Source].[ProductName]				, 		[Source].[ProductCode]				, 		[Source].[ProductGUID]				, 		[Source].[ProductUoMId]				, 		[Source].[FormulaCategoryName]	, 		[Source].[FormulaName]				, 		[Source].[FormulaVersion]			, 		[Source].[FormulaGUID]				, 		[Source].[FormulaId]					, 		[Source].[MRecipeName]				, 		[Source].[MRecipeVersion]			, 		[Source].[MRecipeGUID]				, 		[Source].[MRecipeId]					, 		[Source].[OrderId]					, 		[Source].[PlanEnd]					, 		[Source].[PCellGUID]					, 		[Source].[PCellId]					, 		[Source].[PartId]						, 		[Source].[BatchReleaseGroupId]	, 		[Source].[CreationDateTime]		, 		[Source].[BatchHasNoTags]			, 		[Source].[BatchVersion]				, 		[Source].[LastOSCDateTime]			, 		[Source].[MarkByOSC]					, 		[Source].[BatchTransferTime]		, 		[Source].[InternalSettings] 	) ; end 

################## tr_BatchChangesMerger ##################
 	create trigger [SIMATIC_BATCH].[tr_BatchChangesMerger] on [SIMATIC_BATCH].[BatchChanges] INSTEAD of INSERT 	as 	begin 		set NOCOUNT on; 		merge into [SIMATIC_BATCH].[BatchChanges] as [Target] 		using ( select * from inserted ) as [Source]  		on [Target].OGUID = [Source].OGUID and [Target].EventID = [Source].EventID  		when matched then update set 			[Name]								= [Source].[Name]					, 			[Description]						= [Source].[Description]		, 			[Deleted]							= [Source].[Deleted]				, 			[PredecessorBatchId]				= [Source].[PredecessorBatchId]			, 			[ChainMode]							= [Source].[ChainMode]			, 			[ChainIgnorePredecessorState]	= [Source].[ChainIgnorePredecessorState]		, 			[ActQuantity]						= [Source].[ActQuantity]		, 			[ReleaseTime]						= [Source].[ReleaseTime]		, 			[PlanStart]							= [Source].[PlanStart]			, 			[ActStart]							= [Source].[ActStart]			, 			[ActEnd]								= [Source].[ActEnd]				, 			[StartMode]							= [Source].[StartMode]			, 			[State]								= [Source].[State]				, 			[ExState]							= [Source].[ExState]				, 			[ModificationDateTime]			= [Source].[ModificationDateTime]	, 			[GapTime]							= [Source].[GapTime] 		when not matched then insert values 		( 			[Source].[OGUID]						, 			[Source].[OBJID]						, 			[Source].[EventID]					, 			[Source].[Name]						, 			[Source].[Description]				, 			[Source].[Deleted]					, 			[Source].[PredecessorBatchId]		, 			[Source].[ChainMode]					, 			[Source].[ChainIgnorePredecessorState]				, 			[Source].[ActQuantity]				, 			[Source].[ReleaseTime]				, 			[Source].[PlanStart]					, 			[Source].[ActStart]					, 			[Source].[ActEnd]						, 			[Source].[StartMode]					, 			[Source].[State]						, 			[Source].[ExState]					, 			[Source].[ModificationDateTime]	, 			[Source].[GapTime]				  		); 	end
