----- TRIGGER: CopyBatch  (1690 chars) -----
/* 1) Batch header -> BatchCopy  (verbatim NFM CopyBatch; no offset; safe pattern already present) */
CREATE   TRIGGER [SIMATIC_BATCH].[CopyBatch]
ON [SimaticBatch].[SIMATIC_BATCH].[Batch]
AFTER INSERT, UPDATE
AS
BEGIN
    SET NOCOUNT ON;
    SET XACT_ABORT OFF;                 -- never propagate to the batch engine
    BEGIN TRY
        INSERT INTO [ASMBatchReports].[dbo].[BatchCopy]
            ([ROOTGUID],[ROOTOBJID],[ROOTOTID],[OGUID],[OBJID],[OTID],[Created],[ExAttribute],
             [Quantity],[ProductName],[ProductCode],[ProductGUID],[ProductUoMId],[FormulaCategoryName],
             [FormulaName],[FormulaVersion],[FormulaGUID],[FormulaId],[MRecipeName],[MRecipeVersion],
             [MRecipeGUID],[MRecipeId],[OrderId],[PlanEnd],[PCellGUID],[PCellId],[PartId],
             [BatchReleaseGroupId],[CreationDateTime],[BatchHasNoTags],[BatchVersion],[LastOSCDateTime],
             [MarkByOSC],[BatchTransferTime])
        SELECT
             i.[ROOTGUID],i.[ROOTOBJID],i.[ROOTOTID],i.[OGUID],i.[OBJID],i.[OTID],i.[Created],i.[ExAttribute],
             i.[Quantity],i.[ProductName],i.[ProductCode],i.[ProductGUID],i.[ProductUoMId],i.[FormulaCategoryName],
             i.[FormulaName],i.[FormulaVersion],i.[FormulaGUID],i.[FormulaId],i.[MRecipeName],i.[MRecipeVersion],
             i.[MRecipeGUID],i.[MRecipeId],i.[OrderId],i.[PlanEnd],i.[PCellGUID],i.[PCellId],i.[PartId],
             i.[BatchReleaseGroupId],i.[CreationDateTime],i.[BatchHasNoTags],i.[BatchVersion],i.[LastOSCDateTime],
             i.[MarkByOSC],i.[BatchTransferTime]
        FROM inserted i;
    END TRY
    BEGIN CATCH
        -- swallow: capture must never break the batch engine
    END CATCH;
END;

----- TRIGGER: CopyBatchChanges  (3495 chars) -----
/* 2) BatchChanges -> BatchCopy state fields  (SAFE + SET-BASED rewrite of NFM CopyBatchChanges;
      offsets stripped; keyed on OBJID; never rolls back) */
CREATE   TRIGGER [SIMATIC_BATCH].[CopyBatchChanges]
ON [SimaticBatch].[SIMATIC_BATCH].[BatchChanges]
AFTER INSERT, UPDATE
AS
BEGIN
    SET NOCOUNT ON;
    SET XACT_ABORT OFF;                 -- never roll back the batch engine
    BEGIN TRY
        -- 2a) ensure a header row exists for each completed batch (ActEnd present) that CopyBatch
        --     hasn't already created; pull the full NOT-NULL header from Batch.
        INSERT INTO [ASMBatchReports].[dbo].[BatchCopy]
            ([ROOTGUID],[ROOTOBJID],[ROOTOTID],[OGUID],[OBJID],[OTID],[Created],[ExAttribute],[Quantity],
             [ProductName],[ProductCode],[ProductGUID],[ProductUoMId],[FormulaCategoryName],[FormulaName],
             [FormulaVersion],[FormulaGUID],[FormulaId],[MRecipeName],[MRecipeVersion],[MRecipeGUID],
             [MRecipeId],[OrderId],[PlanEnd],[PCellGUID],[PCellId],[PartId],[BatchReleaseGroupId],
             [CreationDateTime],[BatchHasNoTags],[BatchVersion],[LastOSCDateTime],[MarkByOSC],[BatchTransferTime])
        SELECT
             b.[ROOTGUID],b.[ROOTOBJID],b.[ROOTOTID],b.[OGUID],b.[OBJID],b.[OTID],b.[Created],b.[ExAttribute],b.[Quantity],
             b.[ProductName],b.[ProductCode],b.[ProductGUID],b.[ProductUoMId],b.[FormulaCategoryName],b.[FormulaName],
             b.[FormulaVersion],b.[FormulaGUID],b.[FormulaId],b.[MRecipeName],b.[MRecipeVersion],b.[MRecipeGUID],
             b.[MRecipeId],b.[OrderId],b.[PlanEnd],b.[PCellGUID],b.[PCellId],b.[PartId],b.[BatchReleaseGroupId],
             b.[CreationDateTime],b.[BatchHasNoTags],b.[BatchVersion],b.[LastOSCDateTime],b.[MarkByOSC],b.[BatchTransferTime]
        FROM (SELECT DISTINCT [OBJID] FROM inserted WHERE [ActEnd] IS NOT NULL) i
        JOIN [SimaticBatch].[SIMATIC_BATCH].[Batch] b ON b.[OBJID] = i.[OBJID]
        WHERE NOT EXISTS (SELECT 1 FROM [ASMBatchReports].[dbo].[BatchCopy] bc WHERE bc.[OGUID] = b.[OGUID]);

        -- 2b) update the BatchChanges-sourced fields on the header row(s), KEYED ON OGUID (the unique
        --     batch id) — OBJID is used only for the inherent BatchChanges->Batch link, because OBJID
        --     is not collision-safe in this system (the reason NFM carries the +15000/+28000 offsets).
        UPDATE bc
        SET bc.[Name]                       = i.[Name],
            bc.[Description]                = i.[Description],
            bc.[PredecessorBatchId]        = i.[PredecessorBatchId],
            bc.[ChainMode]                 = i.[ChainMode],
            bc.[ChainIgnorePredecessorState] = i.[ChainIgnorePredecessorState],
            bc.[ActQuantity]               = i.[ActQuantity],
            bc.[ReleaseTime]               = i.[ReleaseTime],
            bc.[PlanStart]                 = i.[PlanStart],
            bc.[ActStart]                  = i.[ActStart],
            bc.[ActEnd]                    = i.[ActEnd],
            bc.[Mode]                      = i.[StartMode],
            bc.[State]                     = i.[State],
            bc.[ExState]                   = i.[ExState]
        FROM [ASMBatchReports].[dbo].[BatchCopy] bc
        JOIN [SimaticBatch].[SIMATIC_BATCH].[Batch] b ON b.[OGUID] = bc.[OGUID]
        JOIN inserted i ON i.[OBJID] = b.[OBJID]
        WHERE i.[ActEnd] IS NOT NULL;
    END TRY
    BEGIN CATCH
        -- swallow: never roll back the batch engine
    END CATCH;
END;

----- TRIGGER: trgInsertOrderDetailsOnOrder  (1102 chars) -----
/* 4) Order -> OrderDetails  (verbatim NFM; safe wrapper added) */
CREATE   TRIGGER [SIMATIC_BATCH].[trgInsertOrderDetailsOnOrder]
ON [SIMATIC_BATCH].[Order]
AFTER INSERT, UPDATE
AS
BEGIN
    SET NOCOUNT ON;
    SET XACT_ABORT OFF;
    BEGIN TRY
        MERGE [ASMBatchReports].[dbo].[OrderDetails] AS target
        USING (SELECT i.ROOTGUID, i.OBJID AS OrderId, i.Name AS OrderName, i.CreationDateTime, i.Quantity, i.Description
               FROM inserted i) AS source
        ON target.ROOTGUID = source.ROOTGUID AND target.OrderId = source.OrderId
        WHEN MATCHED THEN UPDATE SET target.OrderName = source.OrderName, target.CreationDateTime = source.CreationDateTime,
                                     target.Quantity = source.Quantity, target.Description = source.Description
        WHEN NOT MATCHED BY TARGET THEN
            INSERT (ROOTGUID, OrderId, OrderName, CreationDateTime, Quantity, Description)
            VALUES (source.ROOTGUID, source.OrderId, source.OrderName, source.CreationDateTime, source.Quantity, source.Description);
    END TRY
    BEGIN CATCH
    END CATCH;
END;
