CREATE   PROCEDURE dbo.RetrieveAndStoreAllBatchData
  @StartDate DATETIME = NULL,
  @EndDate   DATETIME = NULL
AS
BEGIN
  SET NOCOUNT ON;

  DECLARE @UseDateRange BIT = CASE WHEN @StartDate IS NOT NULL AND @EndDate IS NOT NULL THEN 1 ELSE 0 END;
  DECLARE @LastBT_S1 DATETIME;
  SELECT @LastBT_S1 = LastTimeStamp FROM dbo.DataSyncTracker WHERE [Source Server] = 'Server1';

  IF OBJECT_ID('tempdb..#TempBatchData') IS NOT NULL DROP TABLE #TempBatchData;
  CREATE TABLE #TempBatchData (
    [Source Server] NVARCHAR(50), [Batch GUID] UNIQUEIDENTIFIER, ROOTGUID UNIQUEIDENTIFIER, OrderId INT,
    [Batch Name] NVARCHAR(255), [Product Name] NVARCHAR(255),
    [Batch Act Start] DATETIME, [Batch Act End] DATETIME, [Batch Transfer Time] DATETIME,
    Quantity FLOAT, [Material Name] NVARCHAR(255), [Material Code] NVARCHAR(255),
    sp_prot FLOAT, [SetPoint Float] FLOAT, [Actual Value Float] FLOAT,
    [TimeStamp] DATETIME, FormulaCategoryName NVARCHAR(255)
  );

  ;WITH S1 AS (
    SELECT
      'Server1' AS [Source Server],
      BC.OGUID, BC.ROOTGUID, BC.OrderId,
      BC.Name AS [Batch Name], BC.ProductName AS [Product Name],
      CONVERT(DATETIME, BC.ActStart) AS [Batch Act Start],
      CONVERT(DATETIME, BC.ActEnd)   AS [Batch Act End],
      CONVERT(DATETIME, BC.BatchTransferTime) AS [Batch Transfer Time],
      BC.Quantity,
      PV.sp_matname AS [Material Name],
      PV.sp_matcode AS [Material Code],
      PV.sp_prot,
      PV.sp_float   AS [SetPoint Float],
      PV.av_float   AS [Actual Value Float],
      PV.TimeStamp,
      BC.FormulaCategoryName,
      ROW_NUMBER() OVER (
        PARTITION BY BC.OGUID, BC.FormulaCategoryName, PV.sp_matcode, PV.sp_matname, PV.sp_prot
        ORDER BY PV.TimeStamp DESC
      ) AS rn
    FROM [FAKIEH_SERVER1].ASMBatchReports.dbo.BatchCopy           AS BC
    JOIN [FAKIEH_SERVER1].ASMBatchReports.dbo.ParValueOnline_copy AS PV
      ON BC.OGUID = PV.ROOTGUID
    WHERE PV.av_float > 0 AND PV.sp_float > 0 AND PV.sp_matname IS NOT NULL
      AND (
            (@UseDateRange = 1 AND BC.BatchTransferTime BETWEEN @StartDate AND @EndDate)
         OR (@UseDateRange = 0 AND (@LastBT_S1 IS NULL OR BC.BatchTransferTime > @LastBT_S1))
          )
  )
  INSERT #TempBatchData
  SELECT 'Server1', OGUID, ROOTGUID, OrderId, [Batch Name], [Product Name],
         [Batch Act Start], [Batch Act End], [Batch Transfer Time],
         Quantity, [Material Name], [Material Code], sp_prot,
         [SetPoint Float], [Actual Value Float], TimeStamp, FormulaCategoryName
  FROM S1 WHERE rn = 1;

  ;WITH Agg AS (
    SELECT
      [Source Server], [Batch GUID], [Material Name], [Material Code], FormulaCategoryName, sp_prot,
      MAX(ROOTGUID) AS ROOTGUID, MAX(OrderId) AS OrderId,
      MAX([Batch Name]) AS [Batch Name], MAX([Product Name]) AS [Product Name],
      MIN([Batch Act Start]) AS [Batch Act Start], MAX([Batch Act End]) AS [Batch Act End],
      MAX([Batch Transfer Time]) AS [Batch Transfer Time],
      MAX(Quantity) AS Quantity,
      MAX([SetPoint Float]) AS [SetPoint Float],
      MAX([Actual Value Float]) AS [Actual Value Float]
    FROM #TempBatchData
    GROUP BY [Source Server], [Batch GUID], [Material Name], [Material Code], FormulaCategoryName, sp_prot
  )
  MERGE dbo.BatchMaterials AS T
  USING Agg AS S
    ON  T.[Source Server]     = S.[Source Server]
    AND T.[Batch GUID]        = S.[Batch GUID]
    AND T.[Material Name]     = S.[Material Name]
    AND ((T.[Material Code] IS NULL AND S.[Material Code] IS NULL) OR T.[Material Code] = S.[Material Code])
    AND T.FormulaCategoryName = S.FormulaCategoryName
    AND ((T.sp_prot IS NULL AND S.sp_prot IS NULL) OR T.sp_prot = S.sp_prot)
  WHEN NOT MATCHED THEN
    INSERT ([Source Server],[Batch GUID],ROOTGUID,OrderId,[Batch Name],[Product Name],
            [Batch Act Start],[Batch Act End],[Batch Transfer Time],
            Quantity,[Material Name],[Material Code],sp_prot,[SetPoint Float],[Actual Value Float],FormulaCategoryName)
    VALUES (S.[Source Server],S.[Batch GUID],S.ROOTGUID,S.OrderId,S.[Batch Name],S.[Product Name],
            S.[Batch Act Start],S.[Batch Act End],S.[Batch Transfer Time],
            S.Quantity,S.[Material Name],S.[Material Code],S.sp_prot,S.[SetPoint Float],S.[Actual Value Float],S.FormulaCategoryName)
  WHEN MATCHED THEN
    UPDATE SET
      T.ROOTGUID=S.ROOTGUID, T.OrderId=S.OrderId, T.[Batch Name]=S.[Batch Name], T.[Product Name]=S.[Product Name],
      T.[Batch Act Start]=S.[Batch Act Start], T.[Batch Act End]=S.[Batch Act End],
      T.[Batch Transfer Time]=S.[Batch Transfer Time], T.Quantity=S.Quantity,
      T.[Material Code]=S.[Material Code], T.sp_prot=S.sp_prot,
      T.[SetPoint Float]=S.[SetPoint Float], T.[Actual Value Float]=S.[Actual Value Float],
      T.FormulaCategoryName=S.FormulaCategoryName;

  IF @UseDateRange = 0
  BEGIN
    DECLARE @MaxBT_S1 DATETIME;
    SELECT @MaxBT_S1 = MAX([Batch Transfer Time]) FROM #TempBatchData;
    IF @MaxBT_S1 IS NOT NULL
      MERGE dbo.DataSyncTracker AS T
      USING (SELECT 'Server1' AS Src, @MaxBT_S1 AS BT) AS S
      ON T.[Source Server]=S.Src
      WHEN MATCHED THEN UPDATE SET LastTimeStamp = S.BT
      WHEN NOT MATCHED THEN INSERT([Source Server],LastTimeStamp) VALUES(S.Src,S.BT);
  END

  DROP TABLE #TempBatchData;
END

        bc.FormulaCategoryName   AS [FormulaCategoryName],
        bc.ActStart              AS [Batch Act Start],
        bc.ActEnd                AS [Batch Act End],
        bc.BatchTransferTime     AS [Batch Transfer Time],
        bc.Quantity,

        pv.sp_matname            AS [Material Name],
        pv.sp_matcode            AS [Material Code],
        pv.sp_prot,
        pv.sp_float              AS [SetPoint Float],
        pv.av_float              AS [Actual Value Float],
        pv.[TimeStamp],

        ROW_NUMBER() OVER
        (
            PARTITION BY
                ISNULL(bc.SourceServer,'Unknown'),
                bc.OGUID,
                pv.sp_matcode,
                pv.sp_matname,
                pv.sp_prot
            ORDER BY pv.[TimeStamp] DESC
        ) AS rn

    FROM dbo.BatchCopy bc
    INNER JOIN dbo.ParValueOnline_copy pv
        ON pv.ROOTGUID = bc.OGUID

    WHERE pv.av_float > 0
      AND pv.sp_float > 0
      AND pv.sp_matname IS NOT NULL

      -- Only real batches
      AND bc.FormulaCategoryName IS NOT NULL
      AND ISNULL(bc.[Name],'') <> 'Auto from PV'
),

agg AS
(
    SELECT
        [Source Server],
        [Batch GUID],
        ROOTGUID,
        OrderId,
        [Batch Name],
        [Product Name],
        FormulaCategoryName,
        [Batch Act Start],
        [Batch Act End],
        [Batch Transfer Time],
        Quantity,
        [Material Name],
        [Material Code],
        sp_prot,
        [SetPoint Float],
        [Actual Value Float]
    FROM latest
    WHERE rn = 1
)

MERGE dbo.BatchMaterials AS T
USING agg AS S
    ON  T.[Source Server] = S.[Source Server]
    AND T.[Batch GUID] = S.[Batch GUID]
    AND T.[Material Name] = S.[Material Name]
    AND (
            (T.[Material Code] IS NULL AND S.[Material Code] IS NULL)
            OR T.[Material Code] = S.[Material Code]
        )
    AND (
            (T.sp_prot IS NULL AND S.sp_prot IS NULL)
            OR T.sp_prot = S.sp_prot
        )

WHEN NOT MATCHED THEN
    INSERT
    (
        [Source Server],
        [Batch GUID],
        ROOTGUID,
        OrderId,
        [Batch Name],
        [Product Name],
        FormulaCategoryName,
        [Batch Act Start],
        [Batch Act End],
        [Batch Transfer Time],
        Quantity,
        [Material Name],
        [Material Code],
        sp_prot,
        [SetPoint Float],
        [Actual Value Float]
    )
    VALUES
    (
        S.[Source Server],
        S.[Batch GUID],
        S.ROOTGUID,
        S.OrderId,
        S.[Batch Name],
        S.[Product Name],
        S.FormulaCategoryName,
        S.[Batch Act Start],
        S.[Batch Act End],
        S.[Batch Transfer Time],
        S.Quantity,
        S.[Material Name],
        S.[Material Code],
        S.sp_prot,
        S.[SetPoint Float],
        S.[Actual Value Float]
    )

WHEN MATCHED THEN
    UPDATE SET
        T.ROOTGUID              = S.ROOTGUID,
        T.OrderId               = S.OrderId,
        T.[Batch Name]          = S.[Batch Name],
        T.[Product Name]        = S.[Product Name],
        T.[Batch Act Start]     = S.[Batch Act Start],
        T.[Batch Act End]       = S.[Batch Act End],
        T.[Batch Transfer Time] = S.[Batch Transfer Time],
        T.Quantity              = S.Quantity,
        T.[Material Code]       = S.[Material Code],
        T.sp_prot               = S.sp_prot,
        T.[SetPoint Float]      = S.[SetPoint Float],
        T.[Actual Value Float]  = S.[Actual Value Float],
        T.[FormulaCategoryName] = S.[FormulaCategoryName],
        T.[Source Server]       = S.[Source Server];

END

		WHERE v.[Name] IS NOT NULL
		  AND (
				@LastSync IS NULL
				OR v.[TimeStamp] > @LastSync
			  )
	)

    MERGE dbo.ParValueOnline_copy AS T
    USING S
       ON T.ROOTGUID = S.ROOTGUID
      AND T.[TimeStamp] = S.[TimeStamp]
      AND T.[Name] = S.[Name]

    WHEN NOT MATCHED THEN

            RecHighValue,RecLowValue,
            sp_float,av_float,

            SourceServer

    BEGIN
        MERGE dbo.DataSyncTracker AS T
        USING

            SELECT

        ON T.[Source Server] = S.SourceServer

                LastTimeStamp = S.LastTimeStamp

            INSERT
            (
                [Source Server],
                LastTimeStamp
            )
            VALUES
            (
                S.SourceServer,
                S.LastTimeStamp
            );
    END
